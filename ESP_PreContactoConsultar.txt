SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON

ALTER PROCEDURE [CRM].[ESP_PreContactoConsultar]
(
	 @ProcesoID						INT = NULL	
	,@EmpresaID						INT = NULL
	,@Return						VARCHAR(MAX) OUTPUT 
)
AS
DECLARE
	 @Temp_ProcesoID					INT = @ProcesoID	
	,@SelectCustom						VARCHAR(MAX) = ''
	,@JoinCustom						VARCHAR(MAX) = ''
	,@SQL								NVARCHAR(MAX) = ''
	,@Temp_PreContactoID				INT = NULL
	,@Temp_OrigenPreContactoID			INT = NULL
	,@Temp_FormatoVinculacion			BIT = NULL
	,@Temp_CantidadDecimales			INT = NULL
	,@Temp_SucursalID					INT = NULL	
	,@Temp_EmpresaID					INT = @EmpresaID
BEGIN
	-- ====================================================================================================
	-- SE CARGAN LOS DATOS DEL PRECONTACTO
	-- ====================================================================================================
	SELECT TOP(1)
		 @Temp_PreContactoID = P.PreContactoID
		,@Temp_OrigenPreContactoID = PC.OrigenPrecontactoID
		,@Temp_SucursalID = PC.SucursalID
	FROM
		CRM.Procesos P WITH(NOLOCK)
			INNER JOIN
		CRM.PreContactos PC WITH(NOLOCK)
			ON
				P.PreContactoID = PC.PreContactoID
	WHERE
		P.ProcesoID = @Temp_ProcesoID
			AND
		P.EmpresaID = @Temp_EmpresaID		
		
	-- ====================================================================================================
	-- SI EL ORIGEN DEL PROCESO COMERCIAL DEL PRECONTACTO ES "CAPTACIÓN"
	-- ====================================================================================================
	IF(@Temp_OrigenPreContactoID = 2)
	BEGIN
		EXEC CRM.ESP_PreContactoPropietariosConsultar @ProcesoID = @Temp_ProcesoID, @PreContactoID = @Temp_PreContactoID, @EmpresaID = @Temp_EmpresaID, @Return = @Return OUTPUT
	END
	-- ====================================================================================================
	-- SI EL ORIGEN DEL PROCESO COMERCIAL DEL PRECONTACTO ES "CLIENTES GENERAL"
	-- ====================================================================================================
	ELSE IF(@Temp_OrigenPreContactoID = 3)
	BEGIN
		EXEC CRM.ESP_PreContactoStandardConsultar @ProcesoID = @Temp_ProcesoID, @PreContactoID = @Temp_PreContactoID, @EmpresaID = @Temp_EmpresaID, @Return = @Return OUTPUT
	END
	-- ====================================================================================================
	-- SI EL ORIGEN DEL PROCESO COMERCIAL DEL PRECONTACTO ES "ARRENDATARIOS DE GBI"
	-- ====================================================================================================
	ELSE IF(@Temp_OrigenPreContactoID = 4)
	BEGIN
		EXEC CRM.ESP_PreContactoArrendatariosConsultar @ProcesoID = @Temp_ProcesoID, @PreContactoID = @Temp_PreContactoID, @EmpresaID = @Temp_EmpresaID, @Return = @Return OUTPUT
	END
	-- ====================================================================================================
	-- SI EL ORIGEN DEL PROCESO COMERCIAL DEL PRECONTACTO ES "VENTAS DE GBI"
	-- ====================================================================================================
	ELSE IF(@Temp_OrigenPreContactoID = 5)
	BEGIN
		EXEC CRM.ESP_PreContactoGBIVentasConsultar @ProcesoID = @Temp_ProcesoID, @PreContactoID = @Temp_PreContactoID, @EmpresaID = @Temp_EmpresaID, @Return = @Return OUTPUT
	END
	-- ====================================================================================================
	-- SI EL ORIGEN DEL PROCESO COMERCIAL DEL PRECONTACTO ES "AVALUOS"
	-- ====================================================================================================
	ELSE IF(@Temp_OrigenPreContactoID = 7)
	BEGIN
		EXEC CRM.ESP_PreContactoAvaluosConsultar @ProcesoID = @Temp_ProcesoID, @PreContactoID = @Temp_PreContactoID, @EmpresaID = @Temp_EmpresaID, @Return = @Return OUTPUT
	END
	-- ====================================================================================================
	-- SI EL ORIGEN DEL PROCESO COMERCIAL DEL PRECONTACTO ES "OTHER"
	-- ====================================================================================================
	ELSE
	BEGIN
		-- ====================================================================================================
		-- SE CARGAN LOS CAMPOS PERSONALIZADOS
		-- ====================================================================================================		
		SELECT
			 @SelectCustom += ',[Label' + CPR.NombreColumna + '] = ''' + CP.Nombre + 
							 ''',[Valor' + CPR.NombreColumna + '] = [CPV' + CPR.NombreColumna + '].' + ISNULL(CAST(TD.Descripcion AS VARCHAR(100)), 'Descripcion') + ' '
			,@JoinCustom += ' LEFT JOIN ' + ISNULL(CAST(TD.Tabla AS VARCHAR(100)), 'CRM.CamposPreContactosValores') + ' [CPV' + CPR.NombreColumna + '] ON [CPV' + CPR.NombreColumna + '].' + ISNULL(CAST(TD.LlavePrimaria AS VARCHAR(100)), 'CampoPreContactoValorID') + ' = PC.' + CPR.NombreColumna + ' '
		FROM
			CRM.PreContactos PC WITH(NOLOCK)			
				INNER JOIN
			CRM.CamposPreContactos CP WITH(NOLOCK)
				ON
					PC.OrigenPreContactoID = CP.OrigenPrecontactoID
						AND
					PC.SucursalID = CP.SucursalID
				INNER JOIN
			CRM.CamposPersonalizados CPR WITH(NOLOCK)
				ON
					CP.CampoPersonalizadoID = CPR.CampoPersonalizadoID
				INNER JOIN
			CRM.TiposDatos TD WITH(NOLOCK)
				ON
					CP.TipoDatoID = TD.TipoDatoID
		WHERE
			PC.PreContactoID = @Temp_PreContactoID
				AND
			(
				CP.TipoDatoID >= 50 -- GBI - USO DE INMUEBLE
					OR
				CP.TipoDatoID = 6 -- LISTA DESPLEGABLE
			)

		-- ====================================================================================================
		-- SE CARGA EL FORMATO DE VINCULACIÓN CONFIGURADO PARA STORAGE
		-- ====================================================================================================		
		SELECT TOP(1)
			@Temp_FormatoVinculacion = FormatoVinculacion
		FROM
			STRG.Configuraciones WITH(NOLOCK)
		WHERE
			EmpresaID = @Temp_EmpresaID

		-- ====================================================================================================
		-- SE CONSULTA LA CANTIDAD DE DECIMALES CONFIGURADA
		-- ====================================================================================================		
		SELECT TOP(1)
			@Temp_CantidadDecimales = DecimalesProductos
		FROM
			FAC.Configuraciones WITH(NOLOCK)
		WHERE
			EmpresaID = @Temp_EmpresaID

		-- ====================================================================================================
		-- SE CREA EL QUERY DE CONSULTA
		-- ====================================================================================================	
		SET @SQL = '
			DECLARE
				@Temp_FormatoVinculacionID			INT = NULL

			SELECT TOP(1)					
				@Temp_FormatoVinculacionID = FV.FormatoVinculacionID				
			FROM
				OIK.FormatosVinculacion FV WITH(NOLOCK)
			WHERE
				FV.ProcesoID = ' + CAST(@Temp_ProcesoID AS VARCHAR(10)) + '
					AND
				FV.PreContactoID = ' + CAST(@Temp_PreContactoID AS VARCHAR(10)) + '

			SET @Return = ISNULL((				
				SELECT
					 PC.PreContactoID
					,PC.OrigenPreContactoID
					,PC.Documento
					,PC.Nombres
					,PC.Apellidos
					,PC.NombreCompleto
					,PC.Telefono
					,PC.Celular
					,PC.Email
					,PC.Direccion
					,PC.ClienteTipoPersonaID
					,PC.ClienteTipoDocumentoID
					,PC.ClienteDocumento
					,PC.ClienteResponsabilidadTributariaID
					,PC.ClienteNombres
					,PC.ClienteNombres2
					,PC.ClienteApellidos
					,PC.ClienteApellidos2
					,PC.ClienteNombreCompleto
					,PC.ClienteTelefono
					,PC.ClienteCelular
					,PC.ClienteEmail
					,PC.ClienteEmailFacturacionElectronica
					,PC.ClienteDireccion
					,PC.ClientePaisID
					,PC.ClienteCiudadID							
					,PC.Observaciones							
					,PC.OrigenID
					,PC.Campo1
					,PC.Campo2
					,PC.Campo3
					,PC.Campo4
					,PC.Campo5
					,PC.Campo6
					,PC.Campo7
					,PC.Campo8
					,PC.Campo9
					,PC.Campo10
					,PC.SucursalID
					,PC.PDFCliente
					,P.ProcesoID
					,P.AsesorID
					,P.FormaContactoID
					,P.FormaComoNosConocioID
					,P.FormaComoNosConocioDetalleID
					,P.PalabraBusqueda
					,P.EstadoProcesoID						
					,P.EstadoLineaTiempoID
					,P.FechaCambioLineaTiempo
					,P.FechaCierre
					,P.Fecha
					,P.FechaPosibleServicio
					,P.FechaInviabilidad						
					,P.CausalInviabilidadID
					,P.ObservacionesInviabilidad		
					,P.Color
					,P.UniqueID
					,P.CuentaMensajeriaContactoID
					,IsFormatoVinculacion = ' + CAST(ISNULL(@Temp_FormatoVinculacion, 0) AS VARCHAR(10)) + '
					,CantidadDecimales = ' + CAST(ISNULL(@Temp_CantidadDecimales, 0) AS VARCHAR(10)) + '					
					,OrigenPreContactoNombre = OPC.Nombre
					,FormaContactoNombre = FC.Nombre
					,FormaComoNosConocioNombre = FCNC.Descripcion
					,FormaComoNosConocioDetalleNombre = FCNCD.Nombre
					,EstadoProcesoNombre = EP.Nombre
					,EstadoLineaTiempoNombre = ELT.Nombre
					,CausalInviabilidadNombre = CI.Nombre
					,SucursalNombre = S.Nombre						
					,AsesorNombreCompleto = T.NombreCompleto
					,UsuarioRegistroNombreCompleto = TR.NombreCompleto
					,FormatoVinculacion = JSON_QUERY(ISNULL((
						SELECT TOP(1)					
							 FV.FormatoVinculacionID
							,FV.Consecutivo
							,FV.NombreCompleto
							,FV.EstadoFirmasSignio
							,FV.RutaPDF
							,FV.EnFirmaElectronica
							,FV.TomaSeguro
						FROM
							OIK.FormatosVinculacion FV WITH(NOLOCK)
						WHERE
							FV.FormatoVinculacionID = @Temp_FormatoVinculacionID									
						FOR JSON PATH, INCLUDE_NULL_VALUES, WITHOUT_ARRAY_WRAPPER
					), ''{}''))
					,FormatoSeguro = JSON_QUERY(ISNULL((
						SELECT TOP(1)					
							 FS.FormatoSeguroID
							,FS.Consecutivo
							,FS.ProcesoID
							,FS.EstadoFirmaSignio
							,FS.Ruta
							,FS.EnFirmaElectronica
							,FS.TipoFormatoSeguroID
							,TipoFormatoSeguroNombre = TFS.Nombre
						FROM
							OIK.FormatosSeguros FS WITH(NOLOCK)
								LEFT JOIN
							OIK.TiposFormatosSeguros TFS WITH(NOLOCK)
								ON
									FS.TipoFormatoSeguroID = TFS.TipoFormatoSeguroID
						WHERE
							FS.FormatoVinculacionID = @Temp_FormatoVinculacionID								
						FOR JSON PATH, INCLUDE_NULL_VALUES, WITHOUT_ARRAY_WRAPPER
					), ''{}''))
					' + ISNULL(@SelectCustom, '') + '
					,ProcesosServiciosIniciales = ISNULL((
						SELECT
							 PSI.ProcesoServicioInicialID
							,PSI.TipoProductoID
							,TP.Nombre
						FROM
							CRM.ProcesosServiciosIniciales PSI WITH(NOLOCK)
								INNER JOIN
							PROD.TiposProductos TP WITH(NOLOCK)
								ON
									TP.TipoProductoID = PSI.TipoProductoID
						WHERE
							PSI.ProcesoID = P.ProcesoID
						FOR JSON PATH, INCLUDE_NULL_VALUES
					), ''[]'')			
					,Cotizaciones = ISNULL((
						SELECT
							 CB.CotizacionBodegajeID
							,Aprobado = CAST(CASE WHEN (CB.RequiereAprobacionDescuento = 1 AND CB.EstadoAprobacionDescuentoID = 2 AND CB.Aprobada = 1) OR (CB.RequiereAprobacionDescuento = 0 AND CB.Aprobada = 1) THEN  1 ELSE 0 END AS BIT)
							,CB.Aprobada
							,C.Consecutivo
							,C.UniqueID
							,C.FechaElaboracion
							,C.CotizacionID								
							,ValorCotizacion = ISNULL(C.ValorNeto, 0) + ISNULL(C.ValorImpuestos, 0)
							,CotizacionesVistas = ISNULL((
								SELECT 
									CV.Fecha 
								FROM 
									SIS.CotizacionesVistas CV WITH(NOLOCK)
								WHERE
									CV.CotizacionID = C.CotizacionID
								ORDER BY
									CV.Fecha DESC
								FOR JSON PATH, INCLUDE_NULL_VALUES
							), ''[]'')
						FROM
							SIS.Cotizaciones C WITH(NOLOCK)
								LEFT JOIN
							STRG.CotizacionesBodegaje CB WITH(NOLOCK)
									ON
										C.CotizacionID = CB.CotizacionID									
						WHERE
							C.ProcesoID = P.ProcesoID
						ORDER BY
							C.FechaElaboracion DESC
						FOR JSON PATH, INCLUDE_NULL_VALUES
					), ''[]'')
					,OrdenesServicios = ISNULL((
						SELECT
							 OS.Consecutivo
							,OS.UniqueID
							,OSB.OrdenServicioBodegajeID
							,OS.FechaElaboracion
							,OS.OrdenServicioID																	
							,OS.EstadoOrdenServicioID
							,OS.TerceroID
							,EstadoOrdenServicioNombre = EOS.Nombre
							,ValorOrdenServicio = ISNULL(OS.ValorNeto, 0) + ISNULL(OS.ValorImpuestos, 0)
						FROM								
							SIS.OrdenesServicios OS WITH(NOLOCK)
								LEFT JOIN
							STRG.OrdenesServiciosBodegaje OSB WITH(NOLOCK)
								ON
									OSB.OrdenServicioID = OS.OrdenServicioID										
								LEFT JOIN
							SIS.EstadosOrdenesServicios EOS WITH(NOLOCK)
								ON
									EOS.EstadoOrdenServicioID = OS.EstadoOrdenServicioID
						WHERE
							OS.ProcesoID = P.ProcesoID
						ORDER BY
							OS.FechaElaboracion DESC
						FOR JSON PATH, INCLUDE_NULL_VALUES
					), ''[]'')
					,PreFacturas = ISNULL((
						SELECT
							 TPF.Prefijo
							,PF.Consecutivo
							,PF.PreFacturaID
							,PF.UniqueID
							,PF.FechaCreacion
							,PFT.NombreCompleto
							,ValorPreFactura = ISNULL(PF.ValorNeto, 0) + ISNULL(PF.ValorImpuestos, 0)
						FROM
							SIS.OrdenesServicios OS WITH(NOLOCK)
								INNER JOIN
							STRG.OrdenesServiciosBodegaje OSB WITH(NOLOCK)
								ON
									OSB.OrdenServicioID = OS.OrdenServicioID
								INNER JOIN
							FAC.PreFacturas PF WITH(NOLOCK)
								ON
									PF.OrigenFacturaID = 3
										AND
									PF.CodigoOrigen = OSB.OrdenServicioBodegajeID
								INNER JOIN
							FAC.TiposPreFacturas TPF WITH(NOLOCK)
								ON
									TPF.TipoPreFacturaID = PF.TipoPreFacturaID
								INNER JOIN
							SIS.Terceros PFT WITH(NOLOCK)
								ON
									PF.TerceroID = PFT.TerceroID
						WHERE
							OS.ProcesoID = P.ProcesoID
								AND
							PF.EstadoPrefacturaID = 1
						ORDER BY
							PF.FechaCreacion DESC
						FOR JSON PATH, INCLUDE_NULL_VALUES
					), ''[]'')
					,Facturas = ISNULL((
						SELECT
							 FV.Consecutivo
							,Prefijo = ISNULL(TF.Prefijo + ''-'', '''')
							,FV.UniqueID
							,FV.FacturaID
							,FV.FechaCreacion
							,FV.NombreCompleto
							,FV.ValorFactura
						FROM
						(
							SELECT
								 F.Consecutivo
								,F.UniqueID
								,F.FacturaID
								,F.TipoFacturaID
								,F.FechaCreacion
								,T.NombreCompleto
								,ValorFactura = ISNULL(F.ValorNeto, 0) + ISNULL(F.ValorImpuestos, 0)										
							FROM
								SIS.OrdenesServicios OS WITH(NOLOCK)
									INNER JOIN
								STRG.OrdenesServiciosBodegaje OSB WITH(NOLOCK)
									ON
										OSB.OrdenServicioID = OS.OrdenServicioID
									INNER JOIN
								FAC.Facturas F WITH(NOLOCK)
									ON
										F.OrigenFacturaID = 3
											AND
										F.CodigoOrigen = OSB.OrdenServicioBodegajeID
									INNER JOIN
								SIS.Terceros T WITH(NOLOCK)
										ON
											F.TerceroID = T.TerceroID
							WHERE
								OS.ProcesoID = ' + CAST(@Temp_ProcesoID AS VARCHAR(50)) + '
							UNION
							SELECT
								 F.Consecutivo
								,F.UniqueID
								,F.FacturaID
								,F.TipoFacturaID
								,F.FechaCreacion
								,T.NombreCompleto
								,ValorFactura = ISNULL(F.ValorNeto, 0) + ISNULL(F.ValorImpuestos, 0)
							FROM
								SIS.OrdenesServicios OS WITH(NOLOCK)
									INNER JOIN
								STRG.OrdenesServiciosBodegaje OSB WITH(NOLOCK)
									ON
										OSB.OrdenServicioID = OS.OrdenServicioID
									INNER JOIN
								STRG.Contratos C WITH(NOLOCK)
									ON
										C.OrdenServicioBodegajeID = OSB.OrdenServicioBodegajeID
									INNER JOIN
								FAC.Facturas F WITH(NOLOCK)
									ON
										F.OrigenFacturaID IN(2, 5)
											AND
										F.CodigoOrigen = C.ContratoID
									INNER JOIN
								SIS.Terceros T WITH(NOLOCK)
									ON
										F.TerceroID = T.TerceroID
							WHERE
								OS.ProcesoID = ' + CAST(@Temp_ProcesoID AS VARCHAR(50)) + '
						) AS FV
							LEFT JOIN
						FAC.TiposFacturas TF WITH(NOLOCK)
							ON
								TF.TipoFacturaID = FV.TipoFacturaID
						ORDER BY
							FV.FechaCreacion DESC
						FOR JSON PATH, INCLUDE_NULL_VALUES
					), ''[]'')
					,CountSeguimientos = (
						SELECT
							COUNT(1)
						FROM
							SIS.Seguimientos S WITH(NOLOCK)
						WHERE
							S.OrigenSeguimientoID = ''CRM-PRO''
								AND
							S.OrigenID = P.ProcesoID
					)
					,CountActividades = (
						SELECT
							COUNT(1)
						FROM
							SIS.CalendariosActividades CA WITH(NOLOCK)
						WHERE
							CA.Completada = 0
								AND
							CA.CalendarioActividadOrigenID = 2
								AND
							CA.CodigoOrigen = P.ProcesoID
					)
				FROM
					CRM.PreContactos PC WITH(NOLOCK)
						INNER JOIN
					CRM.Procesos P WITH(NOLOCK)
							ON
								P.PreContactoID = PC.PreContactoID
						LEFT JOIN
					CRM.OrigenesPrecontactos OPC WITH(NOLOCK)
							ON
								PC.OrigenPreContactoID = OPC.OrigenPreContactoID
						LEFT JOIN
					CRM.FormasContactos FC WITH(NOLOCK)
							ON
								FC.FormaContactoID = P.FormaContactoID
						LEFT JOIN
					CRM.FormasComoNosConocio FCNC WITH(NOLOCK)
							ON
								FCNC.FormaComoNosConocioID = P.FormaComoNosConocioID
						LEFT JOIN
					CRM.FormasComoNosConocioDetalles FCNCD WITH(NOLOCK)
							ON
								FCNCD.FormaComoNosConocioDetalleID = P.FormaComoNosConocioDetalleID
						LEFT JOIN
					CRM.EstadosProcesos EP WITH(NOLOCK)
							ON
								EP.EstadoProcesoID = P.EstadoProcesoID
						LEFT JOIN
					CRM.EstadosLineasTiempos ELT WITH(NOLOCK)
							ON
								P.EstadoLineaTiempoID = ELT.EstadoLineaTiempoID
						LEFT JOIN
					CRM.CausalesInviabilidad CI WITH(NOLOCK)
							ON
								CI.CausalInviabilidadID = P.CausalInviabilidadID
						LEFT JOIN
					SIS.Sucursales S WITH(NOLOCK)
							ON
								S.SucursalID = PC.SucursalID							
						LEFT JOIN
					CRM.Asesores A WITH(NOLOCK)
							ON	
								A.AsesorID = P.AsesorID 
						LEFT JOIN
					SIS.Usuarios U WITH(NOLOCK)
							ON	
								U.UsuarioID = A.UsuarioID
						LEFT JOIN
					SIS.Terceros T WITH(NOLOCK)
							ON
								T.TerceroID = U.TerceroID	
						LEFT JOIN
					SIS.Usuarios UR WITH(NOLOCK)
							ON	
								UR.UsuarioID = P.UsuarioID
						LEFT JOIN
					SIS.Terceros TR WITH(NOLOCK)
							ON
								TR.TerceroID = UR.TerceroID	
					' + ISNULL(@JoinCustom, '') + '
				WHERE
					P.ProcesoID = ' + CAST(@Temp_ProcesoID AS VARCHAR(50)) + '
						AND
					PC.PreContactoID = ' + CAST(@Temp_PreContactoID AS VARCHAR(50)) + '
						AND
					PC.EmpresaID = ' + CAST(@Temp_EmpresaID AS VARCHAR(50)) + '				
				FOR JSON PATH, INCLUDE_NULL_VALUES, WITHOUT_ARRAY_WRAPPER
			), ''{}'')
		'

		PRINT(@SQL)
		EXEC sp_executesql @SQL, N'@Return VARCHAR(MAX) OUTPUT', @Return OUTPUT;
	END	
END
